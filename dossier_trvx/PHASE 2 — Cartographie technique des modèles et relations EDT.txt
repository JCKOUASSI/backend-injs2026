from django.apps import apps
from django.db import connection
TARGET_APPS = {
    "edts",
    "presences",
    "formations",
    "scolarite",
}
print("\n" + "=" * 90)
print("AUDIT TECHNIQUE EDT — CARTOGRAPHIE DES MODÈLES")
print("=" * 90)
for model in sorted(
    apps.get_models(),
    key=lambda m: (m._meta.app_label, m.__name__)
):
    if model._meta.app_label not in TARGET_APPS:
        continue
    print(f"\n\nMODEL: {model._meta.label}")
    print(f"TABLE: {model._meta.db_table}")
    for field in model._meta.get_fields():
        if not getattr(field, "concrete", False):
            continue
        relation = ""
        if field.is_relation:
            relation = (
                f" -> {field.related_model._meta.label}"
                if field.related_model else ""
            )
        print(
            f"  {field.name:<35} "
            f"{field.get_internal_type():<20}"
            f"{relation}"
        )
print("\n" + "=" * 90)
print("AUDIT DES VOLUMES DES TABLES EDT")
print("=" * 90)
tables = [
    "edts_executionligne",
    "edts_periodeacademique",
    "edts_revisioncalendrier",
    "edts_verroujour",
]
existing = set(connection.introspection.table_names())
with connection.cursor() as cursor:
    for table in tables:
        if table not in existing:
            print(f"{table}: TABLE ABSENTE")
            continue
        cursor.execute(f'SELECT COUNT(*) FROM "{table}"')
        count = cursor.fetchone()[0]
        print(f"{table}: {count} enregistrements")
print("\n" + "=" * 90)
print("FIN DE L'AUDIT — AUCUNE MODIFICATION EFFECTUÉE")
print("=" * 90)
